uses = ["det-proper-heart-bound-lattice", "det-proper-heart-bound-reduction", "sk-cor-pe-bounded-continuity"]

Let $\Lambda=E$ or $\Qlb$ and $p\colon X\to Y$ proper with geometric fibres of dimension $\leq d$. Then
$R^qp_*M=0$ for all $q>2d+1$ and all $M\in\Cons(X,\Lambda)$.
