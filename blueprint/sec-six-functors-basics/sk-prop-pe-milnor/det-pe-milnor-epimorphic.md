title = "Towers with epimorphic transitions have no derived limit terms"
uses = ["det-pe-milnor-sequence", "det-pe-sections-conservative", "det-pe-sections-limits-colimits", "det-pe-weak-contractible-equivalences"]

Let $(M_n,t_n)$ be a tower of $\mathcal R$-modules on $X_{\mathrm{proét}}$ whose transition
maps $t_n\colon M_{n+1}\to M_n$ are epimorphisms. Then $1-T\colon\prod_nM_n\to\prod_nM_n$ is
an epimorphism, $R^1\varprojlim_nM_n=0$, and $\varprojlim_nM_n\to R\varprojlim_nM_n$ is an
isomorphism in $D(X_{\mathrm{proét}},\mathcal R)$.
