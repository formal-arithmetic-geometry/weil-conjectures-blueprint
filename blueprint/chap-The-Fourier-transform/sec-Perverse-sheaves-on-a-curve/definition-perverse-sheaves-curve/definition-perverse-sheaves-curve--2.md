$$
\begin{aligned}
{}^pD^{\ge0}&=\{K:\mathcal H^aK=0\ (a<-1),\
\mathcal H^{-1}K\text{ has no nonzero punctual subsheaf}\},
\end{aligned}
$$

and ${}^pD^{\le n}={}^pD^{\le0}[-n]$, ${}^pD^{\ge n}={}^pD^{\ge0}[-n]$.
